export const HubUpgradeNotice = () => {
  // The Hub docs bot sets this line when an upgrade proposal passes and clears it when the upgrade is published.
  const upgrade = null;
  if (!upgrade) return null;
  const when = new Date(upgrade.time).toUTCString().replace(':00 GMT', ' UTC');
  return (
    <div
      style={{
        border: '1px solid rgba(214,160,40,0.6)',
        borderRadius: '8px',
        padding: '0.6em 0.9em',
        margin: '0 0 1.2em 0',
        fontSize: '0.9em',
        lineHeight: 1.5,
      }}
    >
      <span>
        Upcoming upgrade: governance proposal{' '}
        <a href={`https://www.mintscan.io/cosmos/proposals/${upgrade.proposal}`}>{upgrade.proposal}</a> upgrades the Hub to
        gaia <a href={`https://github.com/cosmos/gaia/releases/tag/${upgrade.release}`}>{upgrade.release}</a> at
        height {upgrade.height.toLocaleString('en-US')}, estimated {when}. These pages switch to {upgrade.release} when
        the chain reaches that height. See <a href="/hub/latest/hub-tutorials/upgrade-node">Upgrade a node</a>.
      </span>
    </div>
  );
};
